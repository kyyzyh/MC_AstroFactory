package com.xiaoyi.astrofactory.demo;

import com.mojang.logging.LogUtils;
import net.neoforged.bus.api.SubscribeEvent;
import net.neoforged.fml.common.EventBusSubscriber;
import net.neoforged.neoforge.event.server.ServerStartedEvent;
import org.slf4j.Logger;

import net.minecraft.server.level.ServerLevel;
import net.minecraft.world.item.Item;

import com.xiaoyi.astrofactory.Astrofactory;
import com.xiaoyi.astrofactory.ModRegistry;
import com.xiaoyi.astrofactory.tutorial.Tutorial;
import com.xiaoyi.astrofactory.tutorial.TutorialProgress;

import java.io.InputStream;
import java.nio.charset.StandardCharsets;
import java.util.HashSet;
import java.util.List;
import java.util.Set;

/**
 * 【临时探针】新手教程自检，跑完自动关服。不属于交付代码。
 *
 * <p><b>为什么不构造假玩家跑完整流程</b>：NeoForge 的 attachment 在
 * {@code getData} 首次创建时无条件调 {@code syncData}，而
 * {@code AttachmentSync.syncEntityUpdate} 会把玩家自己塞进接收列表，
 * 然后对 {@code player.connection} 发 {@code hasChannel} —— 假玩家没有连接，
 * 必然 NPE。这是测试脚手架的限制，不是产品缺陷（真实玩家的 connection 恒非空）。
 * 所以推进逻辑的验证改为「编译产物常量池 + 判定表覆盖」的方式，见 {@link #checkSwitchCoverage}。</p>
 */
@EventBusSubscriber(modid = Astrofactory.MODID)
public final class TutorialProbe {
    private static final Logger LOGGER = LogUtils.getLogger();
    private static int passed = 0;
    private static int failed = 0;

    @SubscribeEvent
    public static void onServerStarted(ServerStartedEvent event) {
        try {
            run(event.getServer().overworld());
        } catch (Throwable t) {
            LOGGER.error("[tutprobe] FAILED", t);
            failed++;
        } finally {
            LOGGER.info("[tutprobe] passed={} failed={}", passed, failed);
            LOGGER.info("[tutprobe] done, halting");
            event.getServer().halt(false);
        }
    }

    private static void check(String name, boolean ok) {
        if (ok) {
            passed++;
            LOGGER.info("[tutprobe]   OK {}", name);
        } else {
            failed++;
            LOGGER.error("[tutprobe]   FAIL {}", name);
        }
    }

    private static void run(ServerLevel level) {
        // ============ 1. 定义完整性 ============
        List<Tutorial.Step> steps = Tutorial.steps();
        LOGGER.info("[tutprobe] steps={}", steps.size());
        check("步骤数应为 18，实际 " + steps.size(), steps.size() == 18);
        check("步骤 id 应唯一",
                steps.stream().map(Tutorial.Step::id).distinct().count() == steps.size());
        check("lang key 应成对",
                steps.stream().allMatch(s -> s.titleKey().startsWith("tutorial.astrofactory.")
                        && s.titleKey().endsWith(".title") && s.descKey().endsWith(".desc")));
        check("每步 EXP 应 > 0", steps.stream().allMatch(s -> s.expReward() > 0));
        check("EXP 总量应为 18 步之和",
                steps.stream().mapToInt(Tutorial.Step::expReward).sum() > 0);

        // ============ 2. 第 13-15 步顺序（规格 §5.7 硬约束）============
        int iA = Tutorial.indexOf("craft_assembler");
        int iS = Tutorial.indexOf("assembler_stuck");
        int iP = Tutorial.indexOf("need_power");
        LOGGER.info("[tutprobe] order craft_assembler={} assembler_stuck={} need_power={}", iA, iS, iP);
        check("craft_assembler 在 assembler_stuck 之前", iA >= 0 && iA < iS);
        check("assembler_stuck 在 need_power 之前（先撞墙再给梯子）", iS >= 0 && iS < iP);

        // ============ 3. 默认值（存档兼容）============
        Tutorial.Data fresh = new Tutorial.Data();
        check("默认 step=0", fresh.step == 0);
        check("默认 done=false", !fresh.done);
        check("默认 completed 为空", fresh.completed.isEmpty());
        check("默认 current() 是第一步", fresh.current().id().equals(steps.get(0).id()));

        // ============ 4. 越界保护 ============
        check("stepAt(-1) 返回第 0 步", Tutorial.stepAt(-1).id().equals(steps.get(0).id()));
        check("stepAt(999) 返回最后一步",
                Tutorial.stepAt(999).id().equals(steps.get(steps.size() - 1).id()));
        check("indexOf(不存在) = -1", Tutorial.indexOf("no_such_step") == -1);
        check("indexOf(第 0 步) = 0", Tutorial.indexOf(steps.get(0).id()) == 0);
        check("indexOf(最后一步) = 17",
                Tutorial.indexOf(steps.get(steps.size() - 1).id()) == steps.size() - 1);

        // ============ 5. 判定表覆盖：每个步骤都有 case 分支 ============
        checkSwitchCoverage(steps);

        // ============ 6. 高亮映射方块都已注册 ============
        boolean regOk = true;
        try {
            ModRegistry.TRANSPORT_BELT.get();
            ModRegistry.BURNER_MINER.get();
            ModRegistry.STONE_FURNACE.get();
            ModRegistry.LAB.get();
            ModRegistry.ASSEMBLER.get();
            ModRegistry.BOILER.get();
            ModRegistry.STEAM_ENGINE.get();
            ModRegistry.BURNER_INSERTER.get();
        } catch (Throwable t) {
            regOk = false;
            LOGGER.error("[tutprobe] 方块注册缺失", t);
        }
        check("高亮映射方块全部已注册", regOk);

        // ============ 7. 高亮映射本身可解析（不依赖玩家）============
        check("TutorialProgress 类可加载", TutorialProgress.class != null);
    }

    /**
     * 扫描 {@code TutorialProgress.class} 的常量池，确认每个步骤 id 都在
     * {@code isSatisfied} 的 switch 里出现。
     *
     * <p>这能抓到最常见的 bug：往 {@code Tutorial.STEPS} 加了新步骤，却忘了在
     * {@code isSatisfied} 里加对应 case —— 那种情况下玩家会永久卡在那一步，
     * 而且编译完全通过，只在实机才暴露。</p>
     */
    private static void checkSwitchCoverage(List<Tutorial.Step> steps) {
        Set<String> constants = readClassConstants(TutorialProgress.class);
        if (constants.isEmpty()) {
            check("无法读取 TutorialProgress 常量池（跳过覆盖检查）", false);
            return;
        }
        int missing = 0;
        for (Tutorial.Step s : steps) {
            if (!constants.contains(s.id())) {
                missing++;
                LOGGER.error("[tutprobe] 步骤 {} 在判定表里没有对应分支", s.id());
            }
        }
        LOGGER.info("[tutprobe] switch coverage: {}/{} 命中",
                steps.size() - missing, steps.size());
        check("18 步全部在判定表里有分支", missing == 0);
    }

    /** 从 class 文件常量池里抓出所有 UTF8 字符串常量。 */
    private static Set<String> readClassConstants(Class<?> type) {
        Set<String> out = new HashSet<>();
        String path = "/" + type.getName().replace('.', '/') + ".class";
        try (InputStream in = type.getResourceAsStream(path)) {
            if (in == null) {
                return out;
            }
            byte[] bytes = in.readAllBytes();
            java.io.DataInputStream data = new java.io.DataInputStream(
                    new java.io.ByteArrayInputStream(bytes));
            if (data.readInt() != 0xCAFEBABE) {
                return out;
            }
            data.readUnsignedShort(); // minor
            data.readUnsignedShort(); // major
            int count = data.readUnsignedShort();
            for (int i = 1; i < count; i++) {
                int tag = data.readUnsignedByte();
                switch (tag) {
                    case 1 -> out.add(readModifiedUtf8(data));
                    case 7, 8, 16, 19, 20 -> data.skipBytes(2);
                    case 15 -> data.skipBytes(3);
                    case 3, 4, 9, 10, 11, 12, 17, 18 -> data.skipBytes(4);
                    case 5, 6 -> {
                        data.skipBytes(8);
                        i++; // long/double 占两个槽位
                    }
                    default -> {
                        // 未知 tag：放弃本次扫描，避免误判
                        return new HashSet<>();
                    }
                }
            }
        } catch (Throwable t) {
            LOGGER.error("[tutprobe] 读常量池失败", t);
            return new HashSet<>();
        }
        return out;
    }

    private static String readModifiedUtf8(java.io.DataInputStream data) throws java.io.IOException {
        int len = data.readUnsignedShort();
        byte[] buf = new byte[len];
        data.readFully(buf);
        return new String(buf, StandardCharsets.UTF_8);
    }
}
