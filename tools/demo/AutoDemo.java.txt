/**
 * 【临时文件】自动演示 + 自动截图。
 *
 * <p>仅在 run 目录存在 {@code autodemo.flag} 时激活：自动建创造世界 → 铺平台 →
 * 搭"矿机→传送带→熔炉""锅炉→管道→蒸汽机→电线杆→组装机""矿脉+高速带"三个场景 →
 * 依次摆机位截图到 {@code run/screenshots/} → 退出游戏。</p>
 *
 * <p>演示完成后本文件会被删除，不属于交付代码。</p>
 */
@EventBusSubscriber(modid = Astrofactory.MODID, value = Dist.CLIENT)
public final class AutoDemo {
    private static final Logger LOGGER = LogUtils.getLogger();

    private static boolean checked = false;
    private static boolean enabled = false;
    private static volatile boolean buildDone = false;

    private static int phase = 0;
    private static int timer = 0;
    private static int shotIndex = 0;

    private static ServerLevel sLevel;
    private static int ox;
    private static int oz;
    private static int y0;

    private static final List<Shot> SHOTS = new ArrayList<>();

    private AutoDemo() {}

    private record Shot(String name, double x, double y, double z, float yaw, float pitch) {}

    @SubscribeEvent
    public static void onTick(ClientTickEvent.Post event) {
        Minecraft mc = Minecraft.getInstance();
        if (!checked) {
            checked = true;
            File flag = new File(mc.gameDirectory, "autodemo.flag");
            enabled = flag.exists();
            LOGGER.info("[autodemo] flag={} enabled={}", flag.getAbsolutePath(), enabled);
        }
        if (!enabled) {
            return;
        }
        mc.options.pauseOnLostFocus = false;
        try {
            tick(mc);
        } catch (Throwable t) {
            LOGGER.error("[autodemo] failed", t);
            phase = 99;
        }
    }

    private static void tick(Minecraft mc) {
        switch (phase) {
            case 0 -> {
                if (mc.level == null && mc.gui.screen() instanceof TitleScreen) {
                    LOGGER.info("[autodemo] creating world");
                    mc.createWorldOpenFlows().createFreshLevel(
                            "astro_demo",
                            new LevelSettings("AstroDemo", GameType.CREATIVE,
                                    new LevelSettings.DifficultySettings(Difficulty.PEACEFUL, false, false),
                                    true, WorldDataConfiguration.DEFAULT),
                            new WorldOptions(20261120L, false, false),
                            WorldPresets::createNormalWorldDimensions,
                            null);
                    phase = 1;
                    timer = 0;
                }
            }
            case 1 -> {
                if (mc.player != null && mc.getSingleplayerServer() != null && mc.player.tickCount > 80) {
                    // 必须在服务端线程改世界：渲染线程直接 setBlock 会写进客户端
                    // 的世界副本（服务端不认），产线永远不动。
                    var server = mc.getSingleplayerServer();
                    server.execute(() -> {
                        try {
                            build(mc);
                        } catch (Throwable t) {
                            LOGGER.error("[autodemo] build failed", t);
                        } finally {
                            buildDone = true;
                        }
                    });
                    phase = 2;
                    timer = 0;
                }
            }
            case 2 -> {
                if (!buildDone) {
                    return; // 等服务端线程搭完场景再开始计时
                }
                if (timer % 200 == 0) {
                    dumpState(mc);
                }
                // 等产线真跑起来：采矿机 80 tick 一件，机械臂 26 tick 一次，
                // 熔炉配方 200 tick。等 900 tick（45 秒）后再截图，
                // 这样传送带上有在途物品、熔炉在冒火、蒸汽已灌进电线杆。
                if (++timer > 900) {
                    phase = 3;
                    timer = 0;
                    shotIndex = 0;
                    applyShot(mc, 0);
                }
            }
            case 3 -> {
                if (++timer > 30) {
                    grab(mc, SHOTS.get(shotIndex).name());
                    shotIndex++;
                    timer = 0;
                    if (shotIndex >= SHOTS.size()) {
                        phase = 4;
                    } else {
                        applyShot(mc, shotIndex);
                    }
                }
            }
            case 4 -> {
                if (++timer > 80) {
                    LOGGER.info("[autodemo] finished, shutting down");
                    phase = 99;
                    mc.stop();
                }
            }
            default -> {
            }
        }
    }

    // ------------------------------------------------------------ 搭场景

    private static void build(Minecraft mc) {
        var server = mc.getSingleplayerServer();
        sLevel = server.overworld();
        ServerPlayer sp = server.getPlayerList().getPlayers().get(0);

        BlockPos spawn = sLevel.getLevelData().getRespawnData().pos();
        ox = spawn.getX();
        oz = spawn.getZ();

        int maxH = sLevel.getMinY();
        for (int x = ox - 17; x <= ox + 11; x += 2) {
            for (int z = oz - 9; z <= oz + 14; z += 2) {
                maxH = Math.max(maxH, sLevel.getHeight(Heightmap.Types.WORLD_SURFACE, x, z));
            }
        }
        y0 = maxH + 2;
        LOGGER.info("[autodemo] origin=({},{}) y0={}", ox, oz, y0);

        // 铺平台并清空上方
        BlockState stone = Blocks.STONE.defaultBlockState();
        BlockState air = Blocks.AIR.defaultBlockState();
        for (int x = ox - 18; x <= ox + 12; x++) {
            for (int z = oz - 10; z <= oz + 15; z++) {
                sLevel.setBlock(new BlockPos(x, y0 - 1, z), stone, Block.UPDATE_CLIENTS);
                for (int y = y0; y <= y0 + 7; y++) {
                    sLevel.setBlock(new BlockPos(x, y, z), air, Block.UPDATE_CLIENTS);
                }
            }
        }

        buildMinerLine();
        buildSteamPower();
        buildOreAndFastBelt();

        // 玩家：创造飞行 + 快捷栏放模组方块（截图里 HUD 好看）
        sp.setGameMode(GameType.CREATIVE);
        sp.getAbilities().mayfly = true;
        sp.getAbilities().flying = true;
        sp.onUpdateAbilities();
        sp.getInventory().setItem(0, new ItemStack(ModRegistry.TRANSPORT_BELT_ITEM.get()));
        sp.getInventory().setItem(1, new ItemStack(ModRegistry.BURNER_MINER_ITEM.get()));
        sp.getInventory().setItem(2, new ItemStack(ModRegistry.BURNER_INSERTER_ITEM.get()));
        sp.getInventory().setItem(3, new ItemStack(ModRegistry.STONE_FURNACE_ITEM.get()));
        sp.getInventory().setItem(4, new ItemStack(ModRegistry.BOILER_ITEM.get()));
        sp.getInventory().setItem(5, new ItemStack(ModRegistry.STEAM_ENGINE_ITEM.get()));
        sp.getInventory().setItem(6, new ItemStack(ModRegistry.ASSEMBLER_ITEM.get()));
        sp.getInventory().setItem(7, new ItemStack(ModRegistry.POWER_POLE_ITEM.get()));
        sp.getInventory().setItem(8, new ItemStack(ModRegistry.LAB_ITEM.get()));

        buildShots();
    }

    /** 场景 1：采矿机 2×2 → 传送带 → 燃烧臂 → 石熔炉 2×2。 */
    private static void buildMinerLine() {
        Direction f = Direction.SOUTH;
        // 采矿机 master：(ox-6, y0, oz-6)，占 x-6..-7 / z-6..-5
        BlockPos miner = new BlockPos(ox - 6, y0, oz - 6);
        put(sLevel, miner, ModRegistry.BURNER_MINER.get(), f);
        // 下方 2×2 矿脉
        for (int dx = 0; dx < 2; dx++) {
            for (int dz = 0; dz < 2; dz++) {
                BlockPos p = new BlockPos(miner.getX() - dx, y0 - 1, miner.getZ() + dz);
                ore(p, OreVeinBlock.OreKind.IRON, OreVeinBlock.Richness.RICH);
            }
        }
        if (sLevel.getBlockEntity(miner) instanceof BurnerMinerBlockEntity be) {
            be.setItem(BurnerMinerBlockEntity.FUEL_SLOT, new ItemStack(Items.COAL, 64));
        }

        // 出矿口（z = oz-4）两条带
        BlockPos beltA = new BlockPos(ox - 6, y0, oz - 4);
        BlockPos beltB = new BlockPos(ox - 7, y0, oz - 4);
        put(sLevel, beltA, ModRegistry.TRANSPORT_BELT.get(), f);
        put(sLevel, beltB, ModRegistry.TRANSPORT_BELT.get(), f);
        put(sLevel, new BlockPos(ox - 6, y0, oz - 3), ModRegistry.TRANSPORT_BELT.get(), f);
        put(sLevel, new BlockPos(ox - 6, y0, oz - 2), ModRegistry.TRANSPORT_BELT.get(), f);

        // 燃烧臂：(ox-6, y0, oz-1)，背面接带，正面接熔炉
        BlockPos arm = new BlockPos(ox - 6, y0, oz - 1);
        put(sLevel, arm, ModRegistry.BURNER_INSERTER.get(), f);
        if (sLevel.getBlockEntity(arm) instanceof BurnerInserterBlockEntity be) {
            be.getFuelContainer().setItem(0, new ItemStack(Items.COAL, 64));
        }

        // 石熔炉 2×2 master：(ox-6, y0, oz)，占 x-6..-7 / z..z+1
        BlockPos furnace = new BlockPos(ox - 6, y0, oz);
        put(sLevel, furnace, ModRegistry.STONE_FURNACE.get(), f);
        if (sLevel.getBlockEntity(furnace) instanceof StoneFurnaceBlockEntity be) {
            be.setItem(StoneFurnaceBlockEntity.FUEL_SLOT, new ItemStack(Items.COAL, 64));
        }
    }

    /** 场景 2：锅炉 2×3 → 蒸汽管道 → 蒸汽机 3×5 → 电线杆 → 组装机 3×3。 */
    private static void buildSteamPower() {
        Direction f = Direction.SOUTH;

        BlockPos boiler = new BlockPos(ox + 2, y0, oz - 6);
        put(sLevel, boiler, ModRegistry.BOILER.get(), f);
        if (sLevel.getBlockEntity(boiler) instanceof BoilerBlockEntity be) {
            be.setItem(BoilerBlockEntity.FUEL_SLOT, new ItemStack(Items.COAL, 64));
        }

        // 锅炉占 x+1..+2 / z-6..-4；管道沿 x=+2 从 z-3 到 z-1
        for (int z = oz - 3; z <= oz - 1; z++) {
            put(sLevel, new BlockPos(ox + 2, y0, z), ModRegistry.STEAM_PIPE.get(), f);
        }

        // 蒸汽机 master：(ox+2, y0, oz)，占 x+0..+2 / z..z+4
        BlockPos engine = new BlockPos(ox + 2, y0, oz);
        put(sLevel, engine, ModRegistry.STEAM_ENGINE.get(), f);

        // 电线杆：(ox+5, y0, oz+2)，距组装机 master(ox+8,oz) 曼哈顿 5 格，刚好在供电半径内
        put(sLevel, new BlockPos(ox + 5, y0, oz + 2), ModRegistry.POWER_POLE.get(), f);

        // 组装机 master：(ox+8, y0, oz)，占 x+6..+8 / z..z+2
        BlockPos assembler = new BlockPos(ox + 8, y0, oz);
        put(sLevel, assembler, ModRegistry.ASSEMBLER.get(), f);
        if (sLevel.getBlockEntity(assembler) instanceof AssemblerBlockEntity be) {
            // 配方 gcg / .c. / ... ：齿轮 + 电缆 -> 电路板
            be.setItem(0, new ItemStack(ModRegistry.IRON_GEAR.get(), 16));
            be.setItem(1, new ItemStack(ModRegistry.COPPER_CABLE.get(), 16));
            be.setItem(2, new ItemStack(ModRegistry.IRON_GEAR.get(), 16));
            be.setItem(4, new ItemStack(ModRegistry.COPPER_CABLE.get(), 16));
        }
    }

    /** 场景 3：矿脉四种贫富度 + 高速传送带（带上有物品）+ 机械臂。 */
    private static void buildOreAndFastBelt() {
        int z = oz + 6;
        OreVeinBlock.OreKind[] kinds = {
                OreVeinBlock.OreKind.IRON, OreVeinBlock.OreKind.COPPER,
                OreVeinBlock.OreKind.COAL, OreVeinBlock.OreKind.STONE};
        OreVeinBlock.Richness[] rich = {
                OreVeinBlock.Richness.RICH, OreVeinBlock.Richness.NORMAL,
                OreVeinBlock.Richness.POOR, OreVeinBlock.Richness.DEPLETED};
        for (int i = 0; i < 4; i++) {
            ore(new BlockPos(ox - 14 + i, y0, z), kinds[i], rich[i]);
        }

        // 高速带：沿 +X 从 ox-14 到 ox-8
        for (int i = 0; i < 7; i++) {
            BlockPos p = new BlockPos(ox - 14 + i, y0, oz + 8);
            put(sLevel, p, ModRegistry.FAST_TRANSPORT_BELT.get(), Direction.EAST);
        }
        // 带首投料
        if (sLevel.getBlockEntity(new BlockPos(ox - 14, y0, oz + 8)) instanceof TransportBeltBlockEntity be) {
            be.insertAtStart(new ItemStack(Items.RAW_IRON, 1));
            be.insertAtStart(new ItemStack(Items.RAW_COPPER, 1));
        }
        for (int i = 0; i < 3; i++) {
            BlockPos p = new BlockPos(ox - 12 + i, y0, oz + 8);
            if (sLevel.getBlockEntity(p) instanceof TransportBeltBlockEntity be) {
                be.insertAtStart(new ItemStack(Items.RAW_IRON, 1));
            }
        }

        // 普通传送带展示（朝向玩家方向）
        for (int i = 0; i < 4; i++) {
            put(sLevel, new BlockPos(ox - 14 + i, y0, oz + 11), ModRegistry.TRANSPORT_BELT.get(), Direction.EAST);
        }
        put(sLevel, new BlockPos(ox - 10, y0, oz + 10), ModRegistry.BURNER_INSERTER.get(), Direction.NORTH);
    }

    private static void dumpState(Minecraft mc) {
        if (sLevel == null) {
            return;
        }
        var server = mc.getSingleplayerServer();
        if (server == null) {
            return;
        }
        int t = timer;
        // 服务端线程读，避免跨线程读 chunk 拿到 null
        server.execute(() -> {
            StringBuilder sb = new StringBuilder("[autodemo] t=" + t);
            for (int dx = -9; dx <= 1; dx++) {
                for (int dz = -7; dz <= 2; dz++) {
                    BlockPos p = new BlockPos(ox - 6 + dx, y0, oz - 6 + dz);
                    if (!sLevel.isLoaded(p)) {
                        continue;
                    }
                    var be = sLevel.getBlockEntity(p);
                    if (be == null) {
                        continue;
                    }
                    if (be instanceof TransportBeltBlockEntity belt) {
                        sb.append(" belt(").append(dx).append(',').append(dz).append(")=")
                          .append(belt.getItems().size());
                    } else if (be instanceof BurnerMinerBlockEntity miner) {
                        sb.append(" minerFuel=").append(miner.getItem(BurnerMinerBlockEntity.FUEL_SLOT).getCount());
                    } else if (be instanceof BurnerInserterBlockEntity arm) {
                        sb.append(" armHeld=").append(arm.getHeld().getItem()).append(" armFuel=")
                          .append(arm.getFuelContainer().getItem(0).getCount());
                    } else if (be instanceof StoneFurnaceBlockEntity fur) {
                        sb.append(" furIn=").append(fur.getItem(0).getCount())
                          .append(" furOut=").append(fur.getItem(1).getCount())
                          .append(" furFuel=").append(fur.getItem(2).getCount());
                    } else {
                        sb.append(' ').append(be.getClass().getSimpleName());
                    }
                }
            }
            var pole = sLevel.getBlockEntity(new BlockPos(ox + 5, y0, oz + 2));
            if (pole instanceof com.xiaoyi.astrofactory.block.PowerPoleBlockEntity p) {
                sb.append(" pole=").append(p.getEnergy()).append('/').append(p.getEnergyCapacity());
            }
            var eng = sLevel.getBlockEntity(new BlockPos(ox + 2, y0, oz));
            if (eng instanceof com.xiaoyi.astrofactory.block.SteamEngineBlockEntity e) {
                sb.append(" engineSteam=").append(e.getSteam());
            }
            var boi = sLevel.getBlockEntity(new BlockPos(ox + 2, y0, oz - 6));
            if (boi instanceof BoilerBlockEntity b) {
                sb.append(" boilerSteam=").append(b.getSteamAmount());
            }
            var asm = sLevel.getBlockEntity(new BlockPos(ox + 8, y0, oz));
            if (asm instanceof AssemblerBlockEntity a) {
                sb.append(" asmOut=").append(a.getItem(9).getCount());
            }
            String line = sb.toString();
            LOGGER.info(line);
            try {
                java.nio.file.Files.writeString(
                        java.nio.file.Path.of(mc.gameDirectory.getAbsolutePath(), "autodemo_state.log"),
                        line + System.lineSeparator(),
                        java.nio.file.StandardOpenOption.CREATE,
                        java.nio.file.StandardOpenOption.APPEND);
            } catch (Exception ex) {
                LOGGER.warn("[autodemo] write state failed", ex);
            }
        });
    }

    // ------------------------------------------------------------ 机位

    private static void buildShots() {
        SHOTS.clear();
        SHOTS.add(shot("astrofactory_1_overview", ox + 1.5, y0 + 5.0, oz + 10.5, ox + 0.5, y0 + 0.5, oz - 1.5));
        SHOTS.add(shot("astrofactory_2_miner_line", ox - 12.5, y0 + 2.4, oz - 8.5, ox - 6.5, y0 + 0.6, oz - 2.5));
        SHOTS.add(shot("astrofactory_3_steam_power", ox + 9.5, y0 + 3.4, oz - 1.5, ox + 2, y0 + 0.9, oz + 0.5));
        SHOTS.add(shot("astrofactory_4_assembler", ox + 11.0, y0 + 2.4, oz + 4.5, ox + 7, y0 + 0.7, oz + 1));
        SHOTS.add(shot("astrofactory_5_fast_belt", ox - 10.5, y0 + 1.8, oz + 12.0, ox - 11, y0 + 0.3, oz + 8));
        SHOTS.add(shot("astrofactory_6_ore_veins", ox - 12.5, y0 + 1.6, oz + 9.5, ox - 12.5, y0 + 0.3, oz + 6));
        // 机械臂特写：机械臂在 (ox-6, y0, oz-1)，朝南（臂朝 +Z，背面接传送带）
        SHOTS.add(shot("astrofactory_7_inserter", ox - 3.4, y0 + 1.6, oz - 2.8, ox - 6.0, y0 + 0.55, oz - 1.0));
    }

    private static Shot shot(String name, double px, double py, double pz,
                             double tx, double ty, double tz) {
        double dx = tx - px;
        double dy = ty - py;
        double dz = tz - pz;
        float yaw = (float) Math.toDegrees(Math.atan2(-dx, dz));
        double dh = Math.sqrt(dx * dx + dz * dz);
        float pitch = (float) -Math.toDegrees(Math.atan2(dy, dh));
        return new Shot(name, px, py, pz, yaw, pitch);
    }

    private static void applyShot(Minecraft mc, int index) {
        if (sLevel == null) {
            return;
        }
        Shot s = SHOTS.get(index);
        var players = mc.getSingleplayerServer().getPlayerList().getPlayers();
        if (players.isEmpty()) {
            return;
        }
        ServerPlayer sp = players.get(0);
        sp.teleportTo(sLevel, s.x(), s.y(), s.z(), java.util.Set.of(), s.yaw(), s.pitch(), false);
        LOGGER.info("[autodemo] shot {} at ({},{},{})", s.name(), s.x(), s.y(), s.z());
    }

    private static void grab(Minecraft mc, String name) {
        Screenshot.grab(mc.gameDirectory, name + ".png", mc.gameRenderer.mainRenderTarget(), 1,
                component -> LOGGER.info("[autodemo] saved {}", component.getString()));
    }

    // ------------------------------------------------------------ 工具

    private static void put(ServerLevel level, BlockPos pos, Block block, Direction facing) {
        BlockState st = block.defaultBlockState();
        if (st.hasProperty(BlockStateProperties.HORIZONTAL_FACING)) {
            st = st.setValue(BlockStateProperties.HORIZONTAL_FACING,
                    facing.getAxis().isHorizontal() ? facing : Direction.NORTH);
        } else if (st.hasProperty(BlockStateProperties.FACING)) {
            st = st.setValue(BlockStateProperties.FACING, facing);
        }
        level.setBlock(pos, st, Block.UPDATE_ALL);
        BlockState now = level.getBlockState(pos);
        now.getBlock().setPlacedBy(level, pos, now, null, ItemStack.EMPTY);
    }

    private static void ore(BlockPos pos, OreVeinBlock.OreKind kind, OreVeinBlock.Richness rich) {
        BlockState st = ModRegistry.ORE_VEIN.get().defaultBlockState()
                .setValue(OreVeinBlock.KIND, kind)
                .setValue(OreVeinBlock.RICHNESS, rich);
        sLevel.setBlock(pos, st, Block.UPDATE_ALL);
    }
