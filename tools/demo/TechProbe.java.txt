package com.xiaoyi.astrofactory.demo;

import com.mojang.logging.LogUtils;
import net.neoforged.bus.api.SubscribeEvent;
import net.neoforged.fml.common.EventBusSubscriber;
import net.neoforged.neoforge.event.server.ServerStartedEvent;
import org.slf4j.Logger;

import com.xiaoyi.astrofactory.Astrofactory;
import com.xiaoyi.astrofactory.research.ResearchProgress;
import com.xiaoyi.astrofactory.research.Technologies;
import com.xiaoyi.astrofactory.research.Technology;

import java.util.HashSet;
import java.util.List;
import java.util.Set;

/**
 * 【临时探针】科技树自检，跑完自动关服。不属于交付代码。
 */
@EventBusSubscriber(modid = Astrofactory.MODID)
public final class TechProbe {
    private static final Logger LOGGER = LogUtils.getLogger();

    @SubscribeEvent
    public static void onServerStarted(ServerStartedEvent event) {
        try {
            run();
        } catch (Throwable t) {
            LOGGER.error("[techprobe] FAILED", t);
        } finally {
            LOGGER.info("[techprobe] done, halting");
            event.getServer().halt(false);
        }
    }

    private static void run() {
        List<Technology> all = Technologies.all();
        LOGGER.info("[techprobe] total={}", all.size());
        for (Technology t : all) {
            LOGGER.info("[techprobe] node id={} cost={} pre={} unlocks={}",
                    t.id(), t.cost(), t.prerequisites(), t.unlocks());
        }

        // 1) 初始：只有 automation_1 可研究
        Set<String> unlocked = new HashSet<>();
        List<Technology> r0 = Technologies.researchable(unlocked);
        LOGGER.info("[techprobe] step0 researchable={} -> {}",
                r0.size(), r0.stream().map(Technology::id).toList());
        check("step0 size", r0.size() == 1 && r0.get(0).id().equals("automation_1"));

        // 2) 走真实 contribute 路径：automation_1 需要 10 点
        ResearchProgress.Data data = new ResearchProgress.Data();
        boolean unlockedFlag = false;
        for (int i = 0; i < 9; i++) {
            data.contribute("automation_1", 1);
        }
        LOGGER.info("[techprobe] after 9 points: unlocked={} points={}",
                data.isUnlocked("automation_1"), data.getPoints("automation_1"));
        check("not yet unlocked at 9", !data.isUnlocked("automation_1"));

        unlockedFlag = data.contribute("automation_1", 1);
        LOGGER.info("[techprobe] 10th point unlockedFlag={} unlocked={}",
                unlockedFlag, data.isUnlocked("automation_1"));
        check("unlocked at 10", unlockedFlag && data.isUnlocked("automation_1"));

        // 3) 解锁后：electronics / mining_1 / assembly_1 三项同时可研究
        List<Technology> r1 = Technologies.researchable(data.unlocked);
        LOGGER.info("[techprobe] step1 researchable={} -> {}",
                r1.size(), r1.stream().map(Technology::id).toList());
        check("step1 size==3", r1.size() == 3);
        check("step1 contains electronics", r1.stream().anyMatch(t -> t.id().equals("electronics")));
        check("step1 contains mining_1", r1.stream().anyMatch(t -> t.id().equals("mining_1")));
        check("step1 contains assembly_1", r1.stream().anyMatch(t -> t.id().equals("assembly_1")));

        // 4) 前置未满足的不能研究（steam_power 需要 electronics）
        check("steam_power blocked", !Technologies.isResearchable(Technologies.get("steam_power"), data.unlocked));
        check("contribute blocked", !data.contribute("steam_power", 100));

        // 5) 全链路解锁到 steel_processing
        data.contribute("electronics", 10);
        data.contribute("mining_1", 20);
        data.contribute("assembly_1", 20);
        data.contribute("steam_power", 50);
        data.contribute("logistics_1", 50);
        data.contribute("research_1", 50);
        data.contribute("automation_2", 40);
        data.contribute("steel_processing", 50);
        LOGGER.info("[techprobe] fully unlocked={} count={}",
                data.unlocked, data.unlocked.size());
        check("all unlocked", data.unlocked.size() == all.size());

        // 6) 全部解锁后无可研究项
        check("none researchable", Technologies.researchable(data.unlocked).isEmpty());

        // 7) 科研包判定：电路板与红瓶都算
        check("circuit is pack", com.xiaoyi.astrofactory.menu.LabMenu.isResearchPack(
                new net.minecraft.world.item.ItemStack(com.xiaoyi.astrofactory.ModRegistry.ELECTRONIC_CIRCUIT.get())));
        check("red pack is pack", com.xiaoyi.astrofactory.menu.LabMenu.isResearchPack(
                new net.minecraft.world.item.ItemStack(com.xiaoyi.astrofactory.ModRegistry.AUTOMATION_PACK.get())));
        check("dirt is not pack", !com.xiaoyi.astrofactory.menu.LabMenu.isResearchPack(
                new net.minecraft.world.item.ItemStack(net.minecraft.world.item.Items.DIRT)));

        // 8) 红瓶配方存在（工作台 + 组装机）—— 走真实配方表
        var redPackRecipe = com.xiaoyi.astrofactory.recipe.RecipeRegistry.byType(
                        com.xiaoyi.astrofactory.recipe.MachineType.ASSEMBLING)
                .stream()
                .filter(r -> r.result() == com.xiaoyi.astrofactory.ModRegistry.AUTOMATION_PACK.get())
                .toList();
        LOGGER.info("[techprobe] assembler recipes for automation_pack = {}", redPackRecipe.size());
        check("assembler red pack recipe exists", !redPackRecipe.isEmpty());

        LOGGER.info("[techprobe] ALL CHECKS PASSED");
    }

    private static void check(String name, boolean ok) {
        if (ok) {
            LOGGER.info("[techprobe]   OK   {}", name);
        } else {
            LOGGER.error("[techprobe]   FAIL {}", name);
            throw new IllegalStateException("check failed: " + name);
        }
    }
}
