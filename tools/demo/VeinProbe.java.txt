package com.xiaoyi.astrofactory.demo;

import net.minecraft.core.BlockPos;
import net.minecraft.server.level.ServerLevel;
import net.minecraft.world.item.ItemStack;
import net.minecraft.world.level.ChunkPos;
import net.neoforged.bus.api.SubscribeEvent;
import net.neoforged.fml.common.EventBusSubscriber;
import net.neoforged.neoforge.event.server.ServerStartedEvent;

import com.xiaoyi.astrofactory.ModRegistry;
import com.xiaoyi.astrofactory.block.OreVeinBlock;
import com.xiaoyi.astrofactory.world.OreVeinData;

import java.util.HashSet;
import java.util.List;
import java.util.Set;

/**
 * 矿田散布探针。
 *
 * <p>验证的不是「代码不报错」，而是<b>「扩张找矿」这条体验链真的成立</b>：</p>
 * <ol>
 *   <li>{@code shouldScatter} 是确定性纯函数 —— 同输入永远同输出；</li>
 *   <li>散布密度接近 1/18 —— 不是全有也不是全无；</li>
 *   <li>相邻 chunk 的判定不呈周期性（哈希质量）；</li>
 *   <li>出生点保护区内不出散布矿田；</li>
 *   <li>{@code nearestVeins} 返回的距离递增、方位正确；</li>
 *   <li>探矿仪物品已注册、有模型、有配方、语言键可解析。</li>
 * </ol>
 */
@EventBusSubscriber(modid = "astrofactory")
public final class VeinProbe {

    private static final org.slf4j.Logger LOG = com.mojang.logging.LogUtils.getLogger();
    private static int passed = 0;
    private static int failed = 0;

    @SubscribeEvent
    public static void onServerStarted(ServerStartedEvent event) {
        LOG.info("[veinprobe] ===== 矿田散布探针 =====");
        ServerLevel level = event.getServer().overworld();
        try {
            testDeterministic();
            testDensity(level);
            testNoPeriodicity(level);
            testSpawnProtection(level);
            testNearestVeins(level);
            testProspectorItem();
        } catch (Throwable t) {
            LOG.error("[veinprobe] 探针异常", t);
            failed++;
        }
        LOG.info("[veinprobe] passed={} failed={}", passed, failed);
        event.getServer().halt(false);
    }

    // ------------------------------------------------------------ 1. 确定性

    private static void testDeterministic() {
        long seed = 1234567890L;
        ChunkPos c = new ChunkPos(37, -42);
        boolean first = OreVeinData.shouldScatter(c, seed);
        boolean same = true;
        for (int i = 0; i < 100; i++) {
            if (OreVeinData.shouldScatter(c, seed) != first) {
                same = false;
                break;
            }
        }
        check("shouldScatter 确定性（100 次同结果）", same, "结果不稳定");

        // 不同种子应给出不同判定（至少在某些 chunk 上）
        int diff = 0;
        for (int x = 0; x < 50; x++) {
            ChunkPos p = new ChunkPos(x, x * 3 + 1);
            if (OreVeinData.shouldScatter(p, seed) != OreVeinData.shouldScatter(p, seed + 1)) {
                diff++;
            }
        }
        check("换种子会改变判定", diff > 0, "diff=0，种子没起作用");
    }

    // ------------------------------------------------------------ 2. 密度

    private static void testDensity(ServerLevel level) {
        long seed = level.getSeed();
        int total = 0;
        int hits = 0;
        for (int x = -200; x < 200; x++) {
            for (int z = -200; z < 200; z++) {
                total++;
                if (OreVeinData.shouldScatter(new ChunkPos(x, z), seed)) {
                    hits++;
                }
            }
        }
        double ratio = (double) hits / total;
        double expected = 1.0 / OreVeinData.scatterOneIn();
        LOG.info("[veinprobe] 密度 {} / {} = {} (期望 {})", hits, total,
                String.format("%.4f", ratio), String.format("%.4f", expected));
        // 16 万样本，允许 ±30% 相对偏差
        check("散布密度接近 1/" + OreVeinData.scatterOneIn(),
                ratio > expected * 0.7 && ratio < expected * 1.3,
                "实际 " + String.format("%.4f", ratio));
        check("散布不是全有或全无", hits > 0 && hits < total, "hits=" + hits);
    }

    // ------------------------------------------------------------ 3. 无周期性

    private static void testNoPeriodicity(ServerLevel level) {
        long seed = level.getSeed();
        // 检查 x 方向每 18 格是否规律出现（哈希若退化成线性会出现）
        int aligned = 0;
        int hits = 0;
        for (int x = 0; x < 360; x++) {
            if (OreVeinData.shouldScatter(new ChunkPos(x, 77), seed)) {
                hits++;
                if (x % OreVeinData.scatterOneIn() == 0) {
                    aligned++;
                }
            }
        }
        // 若哈希退化，hits 几乎全落在 18 的倍数上
        LOG.info("[veinprobe] x 方向 {} 命中，其中 {} 个落在 18 的倍数", hits, aligned);
        check("命中不集中在周期网格上",
                hits == 0 || aligned < hits, "aligned=" + aligned + " hits=" + hits);

        // 相邻 chunk 不应成片命中（否则矿田会连成一条线）
        int adjacent = 0;
        for (int x = 0; x < 400; x++) {
            if (OreVeinData.shouldScatter(new ChunkPos(x, 3), seed)
                    && OreVeinData.shouldScatter(new ChunkPos(x + 1, 3), seed)) {
                adjacent++;
            }
        }
        LOG.info("[veinprobe] 相邻双命中 {} 次", adjacent);
        check("相邻 chunk 不同时命中", adjacent <= 3, "adjacent=" + adjacent);
    }

    // ------------------------------------------------------------ 4. 出生点保护

    private static void testSpawnProtection(ServerLevel level) {
        var spawn = level.getLevelData().getRespawnData().pos();
        ChunkPos spawnChunk = ChunkPos.containing(spawn);
        var data = level.getDataStorage().computeIfAbsent(OreVeinData.TYPE);
        int guard = OreVeinData.spawnProtectChunks();

        int generated = 0;
        for (int dx = -guard; dx <= guard; dx++) {
            for (int dz = -guard; dz <= guard; dz++) {
                ChunkPos c = new ChunkPos(spawnChunk.x() + dx, spawnChunk.z() + dz);
                if (data.tryGenerateForChunk(level, c, spawnChunk)) {
                    generated++;
                }
            }
        }
        check("出生点保护区内不散布", generated == 0, "生成了 " + generated + " 片");
        LOG.info("[veinprobe] 保护区 {}×{} chunk 检查完毕", guard * 2 + 1, guard * 2 + 1);
    }

    // ------------------------------------------------------------ 5. 探矿仪扫描

    private static void testNearestVeins(ServerLevel level) {
        var spawn = level.getLevelData().getRespawnData().pos();
        ChunkPos spawnChunk = ChunkPos.containing(spawn);
        var data = level.getDataStorage().computeIfAbsent(OreVeinData.TYPE);

        // 出生点保底矿田已由 Astrofactory#onServerStarted 放置
        data.generateIfAbsent(level, spawn);
        check("出生点保底矿田已记录", data.veinChunkCount() >= 4,
                "只有 " + data.veinChunkCount() + " 片");

        // 预测：站在任意位置都该能探到东西（12 chunk 半径内平均约 50 片）
        long seed = level.getSeed();
        List<ChunkPos> near = data.predictVeins(spawnChunk, spawnChunk, 3, 12, seed);
        check("predictVeins 返回非空", !near.isEmpty(), "空列表");
        check("predictVeins 最多返回 3 个", near.size() <= 3, "返回 " + near.size() + " 个");

        // 距离必须递增
        boolean increasing = true;
        long prev = -1;
        for (ChunkPos c : near) {
            int dx = c.x() - spawnChunk.x();
            int dz = c.z() - spawnChunk.z();
            long d2 = (long) dx * dx + (long) dz * dz;
            if (d2 < prev) {
                increasing = false;
            }
            prev = d2;
            LOG.info("[veinprobe] 预测矿田 chunk {} 距离平方 {}", c, d2);
        }
        check("predictVeins 按距离递增", increasing, "顺序不对");

        // 预测结果必须与 shouldScatter 一致（不能骗人）
        int inconsistent = 0;
        for (ChunkPos c : near) {
            if (!OreVeinData.shouldScatter(c, seed)) {
                inconsistent++;
            }
        }
        check("预测结果与实际生成函数一致", inconsistent == 0,
                inconsistent + " 个不一致");

        // 半径 0 应返回空
        ChunkPos farFrom = new ChunkPos(spawnChunk.x() + 10000, spawnChunk.z() + 10000);
        List<ChunkPos> far = data.predictVeins(farFrom, spawnChunk, 3, 0, seed);
        check("半径 0 时返回空", far.isEmpty(), "返回了 " + far.size() + " 个");

        // 预测不能指向出生点保护区（那里不会有散布矿田）
        List<ChunkPos> wide = data.predictVeins(spawnChunk, spawnChunk, 200, 12, seed);
        int inProtection = 0;
        int guard = OreVeinData.spawnProtectChunks();
        for (ChunkPos c : wide) {
            if (Math.abs(c.x() - spawnChunk.x()) <= guard
                    && Math.abs(c.z() - spawnChunk.z()) <= guard) {
                inProtection++;
            }
        }
        check("预测不指向出生点保护区", inProtection == 0, inProtection + " 个落在保护区内");

        // 已生成列表（nearestVeins）也还能用 —— 老接口不能坏
        List<ChunkPos> placed = data.nearestVeins(spawnChunk, 3, 64);
        LOG.info("[veinprobe] 已生成矿田查询返回 {} 个", placed.size());
        check("nearestVeins 仍可用（查已生成）", placed.size() <= 3, "返回 " + placed.size());
    }

    // ------------------------------------------------------------ 6. 探矿仪物品

    private static void testProspectorItem() {
        var item = ModRegistry.PROSPECTOR.get();
        check("探矿仪物品已注册", item != null, "null");
        ItemStack stack = new ItemStack(item);
        String name = item.getName(stack).getString();
        check("探矿仪显示名非键名", !name.equals(item.getDescriptionId()), name);
        check("探矿仪显示名 = 探矿仪/Prospector",
                name.contains("探矿") || name.contains("Prospector"), name);

        for (String res : new String[]{
                "assets/astrofactory/textures/item/prospector.png",
                "assets/astrofactory/models/item/prospector.json",
                "assets/astrofactory/items/prospector.json",
        }) {
            check("探矿仪资源 " + res.substring(res.lastIndexOf('/') + 1),
                    VeinProbe.class.getClassLoader().getResource(res) != null, "缺失");
        }

        // 语言键
        String[] keys = {
                "msg.astrofactory.prospector.header",
                "msg.astrofactory.prospector.line",
                "msg.astrofactory.prospector.none",
                "msg.astrofactory.prospector.hint",
                "dir.astrofactory.north",
                "dir.astrofactory.northeast",
                "dir.astrofactory.east",
                "dir.astrofactory.southeast",
                "dir.astrofactory.south",
                "dir.astrofactory.southwest",
                "dir.astrofactory.west",
                "dir.astrofactory.northwest",
        };
        int missing = 0;
        for (String k : keys) {
            if (net.minecraft.network.chat.Component.translatable(k).getString().equals(k)) {
                missing++;
                LOG.error("[veinprobe] 语言键缺失 {}", k);
            }
        }
        check("探矿仪语言键全部可解析", missing == 0, "缺 " + missing + " 个");

        // 配方存在
        // 由 ContentProbe 覆盖工作台配方检查；此处只验证物品能构造
        check("探矿仪 stacksTo(1)", stack.getMaxStackSize() == 1,
                "实际 " + stack.getMaxStackSize());
    }

    // ------------------------------------------------------------ 工具

    private static void check(String name, boolean ok, String detail) {
        if (ok) {
            passed++;
            LOG.info("[veinprobe] PASS {}", name);
        } else {
            failed++;
            LOG.error("[veinprobe] FAIL {} —— {}", name, detail);
        }
    }
}
