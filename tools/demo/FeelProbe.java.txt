package com.xiaoyi.astrofactory.demo;

import net.minecraft.core.BlockPos;
import net.minecraft.core.registries.BuiltInRegistries;
import net.minecraft.resources.Identifier;
import net.minecraft.server.level.ServerLevel;
import net.neoforged.bus.api.SubscribeEvent;
import net.neoforged.fml.common.EventBusSubscriber;
import net.neoforged.neoforge.event.server.ServerStartedEvent;

import com.xiaoyi.astrofactory.registry.ModBlocks;

/**
 * 手感三件套 + 电力网络性能的验证探针。
 *
 * <p>验证目标（不只是「能编译」）：</p>
 * <ol>
 *   <li>6 个有 ACTIVE 属性的方块都注册了 lightLevel；</li>
 *   <li>lightLevel 在 active=false 时返回 0、active=true 时返回 7；</li>
 *   <li>状态枚举的 6 个 langKey 在 zh_cn / en_us 里都存在；</li>
 *   <li>ElectricNetwork 的偏移表是 515 格（曼哈顿球 r=5）；</li>
 *   <li>ElectricNetwork 同 tick 缓存生效（第二次调用不再扫格子）。</li>
 * </ol>
 */
@EventBusSubscriber(modid = "astrofactory")
public final class FeelProbe {

    private static final org.slf4j.Logger LOG = com.mojang.logging.LogUtils.getLogger();
    private static int passed = 0;
    private static int failed = 0;

    @SubscribeEvent
    public static void onServerStarted(ServerStartedEvent event) {
        LOG.info("[feelprobe] ===== 手感三件套探针 =====");
        try {
            testLightLevel();
            testStatusLang();
            testOffsets();
            testCache();
        } catch (Throwable t) {
            LOG.error("[feelprobe] 探针异常", t);
            failed++;
        }
        LOG.info("[feelprobe] passed={} failed={}", passed, failed);
        event.getServer().halt(false);
    }

    // ------------------------------------------------------------ 1. lightLevel

    private static void testLightLevel() {
        var cases = new Object[][]{
                {"burner_miner", ModBlocks.BURNER_MINER.get(), com.xiaoyi.astrofactory.block.BurnerMinerBlock.ACTIVE},
                {"stone_furnace", ModBlocks.STONE_FURNACE.get(), com.xiaoyi.astrofactory.block.StoneFurnaceBlock.ACTIVE},
                {"assembler", ModBlocks.ASSEMBLER.get(), com.xiaoyi.astrofactory.block.AssemblerBlock.ACTIVE},
                {"boiler", ModBlocks.BOILER.get(), com.xiaoyi.astrofactory.block.BoilerBlock.ACTIVE},
                {"steam_engine", ModBlocks.STEAM_ENGINE.get(), com.xiaoyi.astrofactory.block.SteamEngineBlock.ACTIVE},
                {"offshore_pump", ModBlocks.OFFSHORE_PUMP.get(), com.xiaoyi.astrofactory.block.OffshorePumpBlock.ACTIVE},
        };
        for (Object[] c : cases) {
            String name = (String) c[0];
            var block = (net.minecraft.world.level.block.Block) c[1];
            @SuppressWarnings("unchecked")
            var prop = (net.minecraft.world.level.block.state.properties.BooleanProperty) c[2];

            var off = block.defaultBlockState().setValue(prop, false);
            var on = block.defaultBlockState().setValue(prop, true);
            int lightOff = off.getLightEmission();
            int lightOn = on.getLightEmission();

            check(name + " 熄灯 = 0", lightOff == 0, "实际 " + lightOff);
            check(name + " 亮灯 = 7", lightOn == 7, "实际 " + lightOn);
        }
    }

    // ------------------------------------------------------------ 2. 状态文案

    private static void testStatusLang() {
        String[] keys = {
                "gui.astrofactory.status.running",
                "gui.astrofactory.status.no_power",
                "gui.astrofactory.status.no_input",
                "gui.astrofactory.status.output_full",
                "gui.astrofactory.status.no_fuel",
                "gui.astrofactory.status.idle",
        };
        for (String lang : new String[]{"zh_cn", "en_us"}) {
            String path = "assets/astrofactory/lang/" + lang + ".json";
            try (var in = FeelProbe.class.getClassLoader().getResourceAsStream(path)) {
                if (in == null) {
                    check(lang + " lang 文件存在", false, "读不到 " + path);
                    continue;
                }
                var text = new String(in.readAllBytes(), java.nio.charset.StandardCharsets.UTF_8);
                for (String k : keys) {
                    check(lang + " 有 " + k, text.contains("\"" + k + "\""), "缺失");
                }
            } catch (Exception e) {
                check(lang + " 读取", false, e.toString());
            }
        }
    }

    // ------------------------------------------------------------ 3. 偏移表

    private static void testOffsets() {
        // 曼哈顿球 r=5 的格数公式：(4r^3 + 6r^2 + 8r + 3) / 3
        // r=5: (500 + 150 + 40 + 3)/3 = 693/3 = 231? 不对，用暴力算
        int expect = 0;
        for (int dx = -5; dx <= 5; dx++) {
            for (int dy = -5; dy <= 5; dy++) {
                for (int dz = -5; dz <= 5; dz++) {
                    if (Math.abs(dx) + Math.abs(dy) + Math.abs(dz) <= 5) {
                        expect++;
                    }
                }
            }
        }
        int actual = com.xiaoyi.astrofactory.power.ElectricNetwork.debugOffsetCount();
        check("偏移表格数 = " + expect, actual == expect, "实际 " + actual);
    }

    // ------------------------------------------------------------ 4. 同 tick 缓存

    private static void testCache() {
        var server = net.neoforged.neoforge.server.ServerLifecycleHooks.getCurrentServer();
        if (server == null) {
            check("拿到服务端", false, "null");
            return;
        }
        ServerLevel level = server.overworld();
        BlockPos pos = new BlockPos(0, 100, 0);

        com.xiaoyi.astrofactory.power.ElectricNetwork.debugResetStats();
        com.xiaoyi.astrofactory.power.ElectricNetwork.findNode(level, pos);
        int firstScan = com.xiaoyi.astrofactory.power.ElectricNetwork.debugLastScanCount();

        com.xiaoyi.astrofactory.power.ElectricNetwork.findNode(level, pos);
        int secondScan = com.xiaoyi.astrofactory.power.ElectricNetwork.debugLastScanCount();

        check("首次查询扫了格子", firstScan > 0, "实际 " + firstScan);
        check("同 tick 第二次查询 = 0 次扫描（缓存命中）", secondScan == 0, "实际 " + secondScan);
    }

    // ------------------------------------------------------------ 断言

    private static void check(String what, boolean ok, String detail) {
        if (ok) {
            passed++;
            LOG.info("[feelprobe]   OK {}", what);
        } else {
            failed++;
            LOG.error("[feelprobe]   FAIL {} -> {}", what, detail);
        }
    }
}
