package com.xiaoyi.astrofactory.demo;

import com.mojang.logging.LogUtils;
import net.neoforged.bus.api.SubscribeEvent;
import net.neoforged.fml.common.EventBusSubscriber;
import net.neoforged.neoforge.event.server.ServerStartedEvent;
import org.slf4j.Logger;

import net.minecraft.core.BlockPos;
import net.minecraft.core.Direction;
import net.minecraft.server.level.ServerLevel;
import net.minecraft.world.item.ItemStack;
import net.minecraft.world.item.Items;
import net.minecraft.world.level.block.Block;
import net.minecraft.world.level.block.Blocks;
import net.minecraft.world.level.block.state.BlockState;

import com.xiaoyi.astrofactory.Astrofactory;
import com.xiaoyi.astrofactory.ModRegistry;
import com.xiaoyi.astrofactory.block.TransportBeltBlock;
import com.xiaoyi.astrofactory.block.TransportBeltBlockEntity;

/**
 * 【临时探针】斜坡传送带自检，跑完自动关服。不属于交付代码。
 *
 * 覆盖：
 * 1. 坡度自动判定（上坡 / 下坡 / 平地）
 * 2. 出口格（up 上一格 / down 下一格 / flat 同层）
 * 3. 真实输送：连续 3 个 up 把物品抬高 3 格
 * 4. 真实输送：连续 3 个 down 把物品降低 3 格
 * 5. 背压：坡顶无下游时物品停在末端不丢
 * 6. 存档兼容：默认坡度 flat
 */
@EventBusSubscriber(modid = Astrofactory.MODID)
public final class SlopeProbe {
    private static final Logger LOGGER = LogUtils.getLogger();
    private static int passed = 0;
    private static int failed = 0;

    @SubscribeEvent
    public static void onServerStarted(ServerStartedEvent event) {
        try {
            run(event.getServer().overworld());
        } catch (Throwable t) {
            LOGGER.error("[slopeprobe] FAILED", t);
        } finally {
            LOGGER.info("[slopeprobe] passed={} failed={}", passed, failed);
            LOGGER.info("[slopeprobe] done, halting");
            event.getServer().halt(false);
        }
    }

    private static void check(String name, boolean ok) {
        if (ok) {
            passed++;
            LOGGER.info("[slopeprobe]   OK {}", name);
        } else {
            failed++;
            LOGGER.error("[slopeprobe]   FAIL {}", name);
        }
    }

    private static void clearBox(ServerLevel level, BlockPos center, int r, int yLo, int yHi) {
        for (int dx = -r; dx <= r; dx++) {
            for (int dy = yLo; dy <= yHi; dy++) {
                for (int dz = -r; dz <= r; dz++) {
                    level.setBlock(center.offset(dx, dy, dz), Blocks.AIR.defaultBlockState(), Block.UPDATE_ALL);
                }
            }
        }
    }

    private static void run(ServerLevel level) {
        TransportBeltBlock belt = ModRegistry.TRANSPORT_BELT.get();
        int y0 = 140;

        // ================= 测试 1/2：坡度判定 + 出口格 =================
        BlockPos a0 = new BlockPos(0, y0, 0);
        clearBox(level, a0, 8, -3, 6);
        for (int dx = -3; dx <= 6; dx++) {
            for (int dz = -3; dz <= 3; dz++) {
                level.setBlock(a0.offset(dx, -1, dz), Blocks.STONE.defaultBlockState(), Block.UPDATE_ALL);
            }
        }
        for (int i = 0; i < 3; i++) {
            level.setBlock(a0.offset(i, 0, 0), belt.defaultBlockState()
                    .setValue(TransportBeltBlock.FACING, Direction.EAST), Block.UPDATE_ALL);
        }
        // up 的出口是 front.above()，所以坡顶带子要放在 (3,1,0)，
        // 而 (3,0,0) 必须是实体方块（坡下方要实心，否则玩家看到的是悬空带子）
        level.setBlock(a0.offset(3, 0, 0), Blocks.STONE.defaultBlockState(), Block.UPDATE_ALL);
        level.setBlock(a0.offset(3, 1, 0), belt.defaultBlockState()
                .setValue(TransportBeltBlock.FACING, Direction.EAST), Block.UPDATE_ALL);

        BlockState s2 = level.getBlockState(a0.offset(2, 0, 0));
        LOGGER.info("[slopeprobe] p2 slope={}", s2.getValue(TransportBeltBlock.SLOPE));
        check("末端前有上一层带子应判为 up",
                s2.getValue(TransportBeltBlock.SLOPE) == TransportBeltBlock.Slope.UP);

        if (level.getBlockEntity(a0.offset(2, 0, 0)) instanceof TransportBeltBlockEntity be2) {
            BlockPos exit = be2.exitPos(s2);
            check("up 出口应在上一格", exit.equals(a0.offset(3, 1, 0)));
        } else {
            check("p2 应有 BE", false);
        }

        // ================= 测试 3：连续 3 个 up =================
        BlockPos u0 = new BlockPos(20, y0, 0);
        clearBox(level, u0, 8, -3, 8);
        for (int i = 0; i < 4; i++) {
            level.setBlock(u0.offset(i, i - 1, 0), Blocks.STONE.defaultBlockState(), Block.UPDATE_ALL);
        }
        for (int i = 0; i < 3; i++) {
            level.setBlock(u0.offset(i, i, 0), belt.defaultBlockState()
                    .setValue(TransportBeltBlock.FACING, Direction.EAST)
                    .setValue(TransportBeltBlock.SLOPE, TransportBeltBlock.Slope.UP), Block.UPDATE_ALL);
        }
        BlockPos uTop = u0.offset(3, 3, 0);
        level.setBlock(uTop, belt.defaultBlockState()
                .setValue(TransportBeltBlock.FACING, Direction.EAST)
                .setValue(TransportBeltBlock.SLOPE, TransportBeltBlock.Slope.FLAT), Block.UPDATE_ALL);

        for (int i = 0; i < 3; i++) {
            BlockState st = level.getBlockState(u0.offset(i, i, 0));
            LOGGER.info("[slopeprobe] up[{}] slope={}", i, st.getValue(TransportBeltBlock.SLOPE));
        }

        if (level.getBlockEntity(u0) instanceof TransportBeltBlockEntity beStart) {
            boolean ok = beStart.insertAtStart(new ItemStack(Items.IRON_INGOT, 1));
            check("物品应能投入坡底", ok);
            tickChain(level, u0, 4, true);
            int topCount = countItems(level, uTop);
            LOGGER.info("[slopeprobe] up-chain top items={}", topCount);
            check("连续 3 级 up 应把物品送到 3 格高", topCount == 1);
            check("up 链上不应残留物品", chainCount(level, u0, 4, true) == 1);
        }

        // ================= 测试 4：连续 3 个 down =================
        // 链路：dTop(高层平地) -> down1 -> down2 -> down3 -> dEnd(最低层平地)
        // 每个 down 格都在前格「斜下方」：down[i] 位于 dTop.offset(i, -i, 0)。
        // down 的出口是 front（同层），正好是 down[i+1] —— 串联成立。
        BlockPos dTop = new BlockPos(40, y0 + 3, 0);
        clearBox(level, dTop, 8, -12, 6);
        // 每格下方填石头，避免坡道悬空（仅视觉，不影响逻辑）
        for (int i = 0; i <= 4; i++) {
            int dy = -Math.min(i, 3);
            level.setBlock(dTop.offset(i, dy - 1, 0), Blocks.STONE.defaultBlockState(), Block.UPDATE_ALL);
        }
        level.setBlock(dTop, belt.defaultBlockState()
                .setValue(TransportBeltBlock.FACING, Direction.EAST), Block.UPDATE_ALL);
        for (int i = 1; i <= 3; i++) {
            level.setBlock(dTop.offset(i, -i, 0), belt.defaultBlockState()
                    .setValue(TransportBeltBlock.FACING, Direction.EAST), Block.UPDATE_ALL);
        }
        BlockPos dEnd = dTop.offset(4, -3, 0);
        level.setBlock(dEnd, belt.defaultBlockState()
                .setValue(TransportBeltBlock.FACING, Direction.EAST), Block.UPDATE_ALL);

        // down[1]/down[2]：前方同层是空的、前方下一层有带子 -> 必须判 DOWN
        // down[3]：前方同层正是坡底平带 dEnd -> 按「平接平」判 FLAT（正确）
        for (int i = 1; i <= 2; i++) {
            BlockState st = level.getBlockState(dTop.offset(i, -i, 0));
            LOGGER.info("[slopeprobe] down[{}] slope={}", i, st.getValue(TransportBeltBlock.SLOPE));
            check("down[" + i + "] 应自动判为下坡",
                    st.getValue(TransportBeltBlock.SLOPE) == TransportBeltBlock.Slope.DOWN);
        }
        BlockState d3 = level.getBlockState(dTop.offset(3, -3, 0));
        LOGGER.info("[slopeprobe] down[3] slope={}", d3.getValue(TransportBeltBlock.SLOPE));
        check("down[3] 前方同层是坡底平带，应判 flat",
                d3.getValue(TransportBeltBlock.SLOPE) == TransportBeltBlock.Slope.FLAT);

        BlockPos dFirst = dTop.offset(1, -1, 0);
        if (level.getBlockEntity(dFirst) instanceof TransportBeltBlockEntity beD2) {
            BlockState ds = level.getBlockState(dFirst);
            BlockPos exit = beD2.exitPos(ds);
            // down 的出口在「前方下一格」：那里正是下一级 down 格
            BlockPos want = dFirst.relative(Direction.EAST).below();
            LOGGER.info("[slopeprobe] down exit={} want={}", exit, want);
            check("down 出口应在斜下方（下一级 down 格）", exit.equals(want));
        }

        if (level.getBlockEntity(dTop) instanceof TransportBeltBlockEntity beD) {
            boolean ok = beD.insertAtStart(new ItemStack(Items.GOLD_INGOT, 1));
            check("物品应能投入下坡起点", ok);
            for (int t = 0; t < 400; t++) {
                tickOne(level, dTop);
                for (int i = 1; i <= 3; i++) {
                    tickOne(level, dTop.offset(i, -i, 0));
                }
                tickOne(level, dEnd);
            }
            int endCount = countItems(level, dEnd);
            LOGGER.info("[slopeprobe] down-chain end items={}", endCount);
            check("连续 3 级 down 应把物品送到低 3 格", endCount == 1);
        }

        // ================= 测试 5：背压 =================
        BlockPos lone = new BlockPos(60, y0, 0);
        clearBox(level, lone, 3, -3, 4);
        level.setBlock(lone.below(), Blocks.STONE.defaultBlockState(), Block.UPDATE_ALL);
        level.setBlock(lone, belt.defaultBlockState()
                .setValue(TransportBeltBlock.FACING, Direction.EAST)
                .setValue(TransportBeltBlock.SLOPE, TransportBeltBlock.Slope.UP), Block.UPDATE_ALL);
        if (level.getBlockEntity(lone) instanceof TransportBeltBlockEntity be) {
            be.insertAtStart(new ItemStack(Items.COPPER_INGOT, 1));
            for (int t = 0; t < 100; t++) {
                TransportBeltBlockEntity.serverTick(level, lone,
                        level.getBlockState(lone), be);
            }
            int n = be.getItems().size();
            float prog = n > 0 ? be.getItems().get(0).progress : -1;
            LOGGER.info("[slopeprobe] lone items={} progress={}", n, prog);
            check("坡顶无下游时物品停在末端", n == 1 && prog > 0.8f && prog < 1.0f);
        }

        // ================= 测试 6：默认坡度 flat =================
        BlockState legacy = belt.defaultBlockState().setValue(TransportBeltBlock.FACING, Direction.NORTH);
        check("默认坡度应为 flat",
                legacy.getValue(TransportBeltBlock.SLOPE) == TransportBeltBlock.Slope.FLAT);
    }

    /** 沿链 tick 若干轮。chain 按层递进（up 逐层 +1，down 逐层 -1）。 */
    private static void tickChain(ServerLevel level, BlockPos start, int len, boolean up) {
        for (int t = 0; t < 300; t++) {
            for (int i = 0; i < len; i++) {
                int dy = up ? i : -i;
                BlockPos p = start.offset(i, dy, 0);
                if (level.getBlockEntity(p) instanceof TransportBeltBlockEntity be) {
                    TransportBeltBlockEntity.serverTick(level, p, level.getBlockState(p), be);
                }
            }
        }
    }

    private static void tickOne(ServerLevel level, BlockPos p) {
        if (level.getBlockEntity(p) instanceof TransportBeltBlockEntity be) {
            TransportBeltBlockEntity.serverTick(level, p, level.getBlockState(p), be);
        }
    }

    private static int countItems(ServerLevel level, BlockPos p) {
        if (level.getBlockEntity(p) instanceof TransportBeltBlockEntity be) {
            return be.getItems().size();
        }
        return 0;
    }

    private static int chainCount(ServerLevel level, BlockPos start, int len, boolean up) {
        int n = 0;
        for (int i = 0; i < len; i++) {
            int dy = up ? i : -i;
            n += countItems(level, start.offset(i, dy, 0));
        }
        return n;
    }
}
